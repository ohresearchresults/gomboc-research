def nominated_hull(points):
    import numpy as np
    from scipy.spatial import ConvexHull
    hp=[]
    for p in points:
        d=math.lcm(*(v.denominator for v in p));a=[int(v*d) for v in p]+[d]
        g=math.gcd(*a);hp.append(tuple(v//g for v in a))
    proposal=ConvexHull(np.array([[float(v) for v in p] for p in points])).simplices
    def det(a,b,c):
        return a[0]*(b[1]*c[2]-b[2]*c[1])-a[1]*(b[0]*c[2]-b[2]*c[0])+a[2]*(b[0]*c[1]-b[1]*c[0])
    planes={}
    for tri in proposal:
        mat=[hp[i] for i in tri]
        cof=[(-1)**k*det(*[r[:k]+r[k+1:] for r in mat]) for k in range(4)]
        assert any(cof[:3]);g=math.gcd(*cof);cof=tuple(v//g for v in cof)
        values=[sum(a*b for a,b in zip(cof,p)) for p in hp]
        if max(values)>0:
            assert min(values)>=0
            cof=tuple(-v for v in cof);values=[-v for v in values]
        assert min(values)<0 and max(values)==0
        planes[tuple(i for i,v in enumerate(values) if v==0)]=cof[:3]
    F=[E.cycle(ids,points,n) for ids,n in planes.items()]
    F.sort(key=lambda f:tuple(sorted(f)))
    C.check_surface(points,F)
    return F

def make_core(old,g):

def ceil_fifth(a,D=10**8):
    target=(a.numerator*D**5+a.denominator-1)//a.denominator
    lo=0;hi=1<<((target.bit_length()+4)//5)
    while hi-lo>1:
        mid=(lo+hi)//2
        if mid**5>=target:hi=mid
        else:lo=mid
    v=Q(hi,D)
    assert v**5>=a and Q(hi-1,D)**5<a
    return v
