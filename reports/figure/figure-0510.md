# Polynomial coefficients are in ascending order.
def trim(p):
    p=list(p)
    while len(p)>1 and not p[-1]: p.pop()
    return p

def padd(p,q): return trim([(p[i] if i<len(p) else 0)+(q[i] if i<len(q) else 0) for i in range(max(len(p),len(q)))])
def pneg(p): return [-x for x in p]
def psub(p,q): return padd(p,pneg(q))
def pscale(p,a): return trim([a*x for x in p])
def pmul(p,q):
    z=[F(0)]*(len(p)+len(q)-1)
    for i,x in enumerate(p):
        for j,y in enumerate(q): z[i+j]+=x*y
    return trim(z)

def pdot(a,b):
    z=[F(0)]
    for p,q in zip(a,b): z=padd(z,pmul(p,q))
    return z

def peval(p,t): return sum(x*t**i for i,x in enumerate(p))
def pder(p): return [i*p[i] for i in range(1,len(p))] or [F(0)]
def bern(p,a=F(0),b=F(1)):
    n=len(p)-1
    q=[sum(p[j]*comb(j,k)*a**(j-k)*(b-a)**k for j in range(k,n+1)) for k in range(n+1)]
    return [sum(q[k]*F(comb(i,k),comb(n,k)) for k in range(i+1)) for i in range(n+1)]

def signproof(p,sgn,a=F(0),b=F(1),depth=0):
    if min(sgn*x for x in bern(p,a,b))>0: return [[a,b]]
    assert depth<24,('unproved polynomial sign',a,b,sgn)
    m=(a+b)/2
    return signproof(p,sgn,a,m,depth+1)+signproof(p,sgn,m,b,depth+1)

def featureproof(vals,want,a=F(0),b=F(1),depth=0):
    bounds=[bern(p,a,b) for p in vals]
    if want and all(min(z)>0 for z in bounds): return [[a,b,'all']]
    if not want:
        for i,z in enumerate(bounds):
            if max(z)<0: return [[a,b,i]]
    assert depth<24,('unproved feature',a,b,want)
    m=(a+b)/2
    return featureproof(vals,want,a,m,depth+1)+featureproof(vals,want,m,b,depth+1)

# Optional numerical producers. Neither is needed to recover the literal rays.
def numeric_bars(P,E,np):
    R=np.sum(P*P,axis=1); E=np.asarray(E,dtype=int)
    d=P[E[:,1]]-P[E[:,0]]
    t=np.clip(-np.sum(P[E[:,0]]*d,axis=1)/np.sum(d*d,axis=1),0,1)
    W=np.sum((P[E[:,0]]+t[:,None]*d)**2,axis=1)
    uf=list(range(len(P))); result=[]
    def find(i):
        while uf[i]!=i: uf[i]=uf[uf[i]];i=uf[i]
        return i
    for k in sorted(range(len(E)),key=lambda k:(W[k],*E[k]),reverse=True):
        a,b=map(find,E[k])
        if a==b: continue
        if (R[a],-a)<(R[b],-b): a,b=b,a
        gap=np.log(R[b]/W[k]) if W[k]>0 else np.inf
        if gap>1e-12: result.append(float(gap))
        uf[b]=a
    return result

## Continuation

The useful next entrance is the literal H23 t=1 or MONO44 t=1 endpoint, not a replay of the generic rank test. Both already have a finite actual-mass descent certificate for the entire unwanted physical basin. The immediate obstruction on the displayed H23 ray is a cluster of mechanical births well before the first support-cell boundary, not loss of the five carriers or lack of centroid controllability.

A concrete next change is to permit a coherent exchange of the nearly active polar witnesses rather than extending the same guarded ray. In H23, frozen input rows 2 and 13 are especially informative because they become additional stable faces while their common protected carrier remains fixed. Releasing those planes together with their affected carrier, while using the remaining mass controls and auditing the whole polar basin, is a genuine new architecture to test. It is not certified sufficient. The current experiment establishes finite protected actual-mass motion, whole-bar improvement, and the subsequent birth/flip mechanisms; it does not establish cancellation of the remaining physical basin.
