## Что остаётся доказанным и что остаётся открытым

Прежнее препятствие охватывает `K(h)=conv(W,(0,h,0))` при **фиксированном** W, евклидовой метрике и всех действительных h>max_WY. В этой области S≥3 при фактическом изменяющемся объёмном центре. Для K16 сохраняющийся противосвидетель имеет исходные вершины[0,24,23], нормаль(−6399010,−15517405,32043131), поддержку323472120441/10000 и доказанную нижнюю границу центра−21/50. **Нижние носители, BASE, неосевое движение вершины, другие метрические и материальные изменения этим не исключены.** :contentReference[oaicite:5]{index=5} :contentReference[oaicite:6]{index=6}

Полезнейший следующий конструктивный шаг не изменился: совместная перестройка нижнего корпуса и BASE с фактической массой и всем29/31 Q-интерфейсом, с восстановлением физического положительного леса после событий оболочки. Изменение одной грани[0,24,23] может разрушить данный противосвидетель, но не является доказательством отмены устойчивого пакета. В этом продолжении такая перестройка **не доведена до проверенного конечного тела**.

## Полный воспроизводимый код единственной новой проверки

Следующий исходник воспроизводит именно проверку упаковки и обнаружение внешнего SHA-расхождения. Он не запускает геометрический поиск, не пересчитывает массы и не приписывает сохранённым отчётам свежий механический аудит.

Зависимости: CPython3.13.5, стандартная библиотека. Точные входы — два уже переданных файла с указанными выше размерами и SHA. Сохранить как `CHECK_RA37_CLOSEOUT.py`, UTF-8/LF. Наблюдавшийся размер2936байт, SHA-256 `ca53c7ba534fccc3477521f221e294b5c764ea7fc4df922098032352fd0c422f`.

```python
"""RA37 inherited-source integrity only; no geometry or mechanics execution.
Observed environment: CPython 3.13.5, standard library.
Usage: python -B -S CHECK_RA37_CLOSEOUT.py JOURNAL ZIP OUT
"""
import ast
import base64
import hashlib
import json
import lzma
from pathlib import Path
import sys
import zipfile


def sha(data):
    return hashlib.sha256(data).hexdigest()


def main():
    if len(sys.argv) != 4:
        raise SystemExit('Usage: CHECK_RA37_CLOSEOUT.py JOURNAL ZIP OUT')
    journal = Path(sys.argv[1]).read_bytes()
    archive = Path(sys.argv[2])
    out = Path(sys.argv[3])
    assert len(journal) == 340512
    assert sha(journal) == '12b5d86fe7c9c0ac72d6e92d3f6c9e3abfa0ca202c0c206f1b2bd364e76f4dcc'
    assert len(archive.read_bytes()) == 1058767
    assert sha(archive.read_bytes()) == '5040d872652a84f69f5042867d2d2a8fea69bd1f7bb60ba623e31b1dc0d8f439'
    section = journal.decode('utf-8').split('## Parent [internal-source-reference]-\n', 1)[1]
    payload = section[section.index('# RA37 —'):].encode('utf-8')
    assert len(payload) == 50105
    assert sha(payload) == '4cceaf1344bf1d0cff8d77be9a4e8fc601e45dc9c42189ed713a7ae67f89ce23'
    advertised = 'baa91124c0ffe0b44dbafd069c5a50594083776822b2c0391cc7fbf0213ecc46'
    assert sha(payload) != advertised
    code = payload.decode('utf-8').split('```python\n', 1)[1].split('\n```', 1)[0] + '\n'
    assert len(code.encode()) == 19742
    assert sha(code.encode()) == 'e97c01cdd644bdb2d54596018d5de9505654398cb051fc606974d1810d68296d'
    nodes = ast.parse(code).body
    data = ast.literal_eval(next(n.value for n in nodes if isinstance(n, ast.Assign)
        and any(isinstance(t, ast.Name) and t.id == 'DATA' for t in n.targets)))
    raw = lzma.decompress(base64.b64decode(data, validate=True))
    assert len(raw) == 37100
    assert sha(raw) == 'e5a87a0a414de552dc4f443b528114a10c92d0f31e4c17807d54b404829a3319'
    sources = json.loads(raw)
    assert len(sources) == 6
    out.mkdir(parents=True, exist_ok=True)
    with zipfile.ZipFile(archive) as z:
        for name, text in sorted(sources.items()):
            assert Path(name).name == name
            b = text.encode('utf-8')
            assert z.read('RA37/' + name) == b
            (out / name).write_bytes(b)
        expected = json.loads(sources['EXPECTED.json'])
        assert len(expected) == 21
        for name, receipt in sorted(expected.items()):
            b = z.read('RA37/reports/' + name)
            assert len(b) == receipt['bytes'] and sha(b) == receipt['sha256']
    (out / 'RECOVER_RA37.py').write_bytes(code.encode('utf-8'))
    print('RA37_PARENT_SERIALIZATION_MISMATCH_REPORTED', len(payload), sha(payload))
    print('RA37_INHERITED_SOURCE_DICT_AND_SIX_ZIP_UNITS_MATCH_PASS', len(sources))
    print('RA37_INHERITED_STORED_REPORT_RECEIPTS_PASS', len(expected))
    print('RA37_CLOSEOUT_NO_NEW_GEOMETRY_OR_MECHANICAL_REPLAY')


if __name__ == '__main__':
    main()
```
