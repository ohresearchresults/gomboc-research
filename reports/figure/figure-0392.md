def seed_tree(Z,E,root):
    W=[]
    for i,x in enumerate(Z):
        choices=[]
        for j in sorted(E[i]):
            d=sub(Z[j],x); q=dot(x,d)
            if q>0: choices.append((q*q/(dot(x,x)*dot(d,d)),j))
        if choices: W.append([i,max(choices)[1]])
    assert len(W)==34
    parent=dict(W)
    for i in parent:
        seen=set()

            assert i not in seen; seen.add(i); i=parent[i]
    return W

def solve(cols,rhs):
    n=len(rhs); m=len(cols)
    A=[[F(cols[j][i]) for j in range(m)]+[F(rhs[i])] for i in range(n)]
    piv=[]; r=0
    for j in range(m):
        eligible=[i for i in range(r,n) if A[i][j]]
        if not eligible: continue
        k=eligible[0]; A[r],A[k]=A[k],A[r]
        q=A[r][j]; A[r]=[x/q for x in A[r]]
        for i in range(n):
            if i!=r:
                q=A[i][j]; A[i]=[A[i][k]-q*A[r][k] for k in range(m+1)]
        piv.append(j); r+=1
    assert len(piv)==m and all(any(v[:-1]) or v[-1]==0 for v in A)
    x=[F(0)]*m
    for i,j in enumerate(piv): x[j]=A[i][-1]
    assert all(sum(cols[j][i]*x[j] for j in range(m))==rhs[i] for i in range(n))
    return x

def terminal_components(graph):
    reach=[]
    for i in range(len(graph)):
        seen={i}; todo=[i]
        while todo:
            for j in graph[todo.pop()]:
                if j not in seen: seen.add(j); todo.append(j)
        reach.append(seen)
    unseen=set(range(len(graph))); ans=[]
    while unseen:
        i=min(unseen); c={j for j in reach[i] if i in reach[j]}
        unseen-=c
        if all(j in c for i in c for j in graph[i]): ans.append(sorted(c))
    return ans

def bounds_and_cuts(X,P,W,bases,expected):
    C=rows(X,W); lo=[None]*6; hi=[None]*6; used=set(); duals=[]
    for k in range(6):
        for j,s in enumerate([1,-1]):

            z=solve([C[i] for i in ids]+[T],rhs)

            used.update(i for i,q in zip(ids,z[:-1]) if q>0 and i<len(W))
            mu=z[-1]
            if s==1: lo[k]=F(math.floor(mu*10**6),10**6)
            else: hi[k]=F(math.ceil(-mu*10**6),10**6)
            duals.append(dict(coordinate=k,sign=s,basis=ids,coefficients=z))
    B=[max(abs(a),abs(b)) for a,b in zip(lo,hi)]
    delta=lo[0]*lo[1]-B[3]**2; assert delta>0
    kh=[hi[1]*hi[2]/delta,hi[0]*hi[2]/delta,F(1),
        (B[4]*B[5]+B[2]*B[3])/delta,
        (B[3]*B[5]+B[1]*B[4])/delta,
        (B[3]*B[4]+B[0]*B[5])/delta]
    kl=[F(0),F(0),F(1),-kh[3],-kh[4],-kh[5]]
    graph=[[] for _ in P]; upper={}
    for i,p in enumerate(P):
        for j,q in enumerate(P):
            if i==j: continue
            u=sum(t*(b if t>=0 else a)
                  for t,a,b in zip(coeff(p,sub(q,p)),kl,kh))
            upper[i,j]=u
            if u>=0: graph[i].append(j)
    cuts=terminal_components(graph)
    assert cuts==expected and len(cuts)==2
    proofs=[]
    for c in cuts:
        u,i,j=max((upper[i,j],i,j) for i in c for j in range(len(P)) if j not in c)
        assert u<0
        proofs.append(dict(ids=c,maximum_outgoing_upper=u,tight_pair=[i,j]))
    return dict(Gram_box=list(zip(lo,hi)),inverse_box=list(zip(kl,kh)),
                delta_lower=delta,duals=duals,

def lp(q,A,equalities):
    q=np.array(q,float); q/=np.max(abs(q))
    return linprog(q,A_ub=-A,b_ub=np.zeros(len(A)),A_eq=equalities,
                   b_eq=[1]+[0]*(len(equalities)-1),
                   bounds=[(None,None)]*6,method='highs')
def exact_dual(q,C,eqs,result):
    assert result.success
    ids=np.flatnonzero(abs(result.ineqlin.marginals)>1e-9).tolist()
    z=solve([C[i] for i in ids]+eqs,q)
    assert all(x>=0 for x in z[:len(ids)])
    return ids,z

def export_orbit(name,X,P,E,FE,edges,path):
