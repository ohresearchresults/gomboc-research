    for f, face in enumerate(FACES):
        n, h = PLANES[f]
        a,b,c = (Y[i] for i in face[:3])
        cross = (b-a).cross(c-a)
        assert same(cross.cross(n), S.zeros(3,1))
        proofs[f'orientation:{f}'] = certify_sign(cross.dot(n))
        for i,z in enumerate(Y):
            slack = S.expand(h-n.dot(z))
            if i in face:
                assert slack == 0
            else:
                proofs[f'support:{f}:{i}'] = certify_sign(slack)
        for k,i in enumerate(face):
            nxt = Y[face[(k+1)%len(face)]]-Y[i]
            prv = Y[face[(k-1)%len(face)]]-Y[i]
            proofs[f'turn:{f}:{i}'] = certify_sign(n.dot(nxt.cross(prv)))
        height = S.expand(h-n.dot(g))
        hs.append(height)
        proofs[f'height:{f}'] = certify_sign(height)
    bd,bj = boundary_moments(Y)
    assert bd == D and same(bj,J)
    # Independent positive volume fan with apex at the true vertex 7.
    o = Y[7]
    fanD = S.S(0)
    fanJ = S.zeros(3,1)
    cells = []
    for face in FACES:
        if 7 in face:
            continue
        for k in range(1,len(face)-1):
            ids = (7,face[0],face[k],face[k+1])
            a,b,c = (Y[i] for i in ids[1:])
            dv = S.expand((a-o).dot((b-o).cross(c-o)))
            proofs['fan:'+str(ids)] = certify_sign(dv)
            fanD += dv
            fanJ += dv*(o+a+b+c)/4
            cells.append(ids)
    assert len(cells) == 9 and zero(fanD-D) and same(fanJ,J)
    assert zero(D/6-R(15,64))
    return hs, cells, proofs


def primitive_checks(hs):
    pp = {}
    qq = {}
    signsP = {}
    signsQ = {}
    X = [z-g for z in Y]
    for i,j in ARROWS:
        f = S.expand((X[i].T*H*(Y[j]-Y[i]))[0])
        sign = 1 if (i,j) in POS_P else -1
        pp[f'{i}>{j}'] = certify_sign(f,sign)
        signsP[i,j] = sign
    for i,j in DARROWS:
        n, unused = PLANES[i]
        m, unused = PLANES[j]
        f = S.expand(hs[i]*(n.T*K*m)[0]-hs[j]*(n.T*K*n)[0])
        sign = 1 if (i,j) in POS_Q else -1
        qq[f'{i}>{j}'] = certify_sign(f,sign)
        signsQ[i,j] = sign
    U = [i for i in range(10)
         if all(signsP[a,b]<0 for a,b in ARROWS if a==i)]
    stable = [i for i in range(9)
              if all(signsQ[a,b]<0 for a,b in DARROWS if a==i)]
    saddles = []
    for i,j in EDGES:
        f,h = EF[i,j]
        if max(signsP[i,j],signsP[j,i],signsQ[f,h],signsQ[h,f])<0:
            saddles.append((i,j))
    assert U == [2,7]
    assert stable == [0,3,4,5,6]
    assert saddles == [(1,2),(2,8),(3,9),(4,9),(5,6)]
    assert len(stable)+len(U)-len(saddles) == 2

    return pp,qq,{'S':stable,'U':U,'H':saddles,

def integer_table_checks(hs):

def transport_checks():
    A = S.diag(e,1,1)
    raw = tuple(A*z for z in Y)
    rawg = A*g
    rawD = e*D
    rawJ = e*A*J
    M = (e**2*A.inv().T*H*A.inv()).applyfunc(S.cancel)
    N = (A*Hinv*A.T/e**2).applyfunc(S.cancel)
    assert same(M*N,S.eye(3))
    expectedM = S.Matrix(((1001,3000*e,-2000*e),
                         (3000*e,9001*e**2,-6000*e**2),
                         (-2000*e,-6000*e**2,4001*e**2)))/1000
    assert same(M,expectedM)
    trace = S.trace(M)
    assert zero(trace-(1001+13002*e**2)/1000)
    bd,bj = boundary_moments(raw)
    assert zero(bd-rawD) and same(bj,rawJ)
    assert same(rawg,S.Matrix((e*(757-128*e)/2880,
                              (933-264*e+128*e**2)/2880,
                              (61+16*e)/180)))
    for i,j in ARROWS:
        qy = -sym(D*Y[i]-J,Y[j]-Y[i])
        qr = -sym(rawD*raw[i]-rawJ,raw[j]-raw[i])
        assert same(qr,e*A*qy*A.T)
        dr = ((raw[i]-rawg).T*M*(raw[j]-raw[i]))[0]
        dy = ((Y[i]-g).T*H*(Y[j]-Y[i]))[0]
        assert zero(dr-e**2*dy)
    M0 = (M/trace).subs(e,0)
    assert M0 == S.diag(1,0,0)
    for i,j in ARROWS:
        a = (raw[i]-rawg).subs(e,0)
        edge = (raw[j]-raw[i]).subs(e,0)
        assert (a.T*M0*edge)[0] == 0
    tree = ((0,7),(1,0),(2,1),(3,2),(4,3),
            (5,4),(6,7),(8,7),(9,7))
    assert all(a in ARROWS for a in tree)
    parent = dict(tree)
    assert set(parent) == set(range(10))-{7}
    for start in parent:
        seen = set()
        here = start

            assert here not in seen
            seen.add(here)
            here = parent[here]
    assert R(14001,1000)*R(9,4) < 32
    assert R(45,32)*32 == 45
    return {'raw_vertices':raw,'raw_volume':rawD/6,'raw_centroid':rawg,
            'M':M,'literal_inverse':N,'metric_trace':trace,

            'tree_feasible_for_positive_epsilon':False,
            'all_raw_boundary_primitives_zero':True,

def enc(value):
    if isinstance(value,S.MatrixBase):
        return [[enc(x) for x in row] for row in value.tolist()]
    if isinstance(value,S.Basic):
        return str(value)
    if isinstance(value,dict):
        return {str(k):enc(v) for k,v in value.items()}
    if isinstance(value,(tuple,list)):
        return [enc(x) for x in value]
    return value
