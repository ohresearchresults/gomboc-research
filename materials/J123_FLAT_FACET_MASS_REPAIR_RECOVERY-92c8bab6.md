"""Finite actual-mass physical-forest repair on an exact flat-facet manifold.
The nominated fourth vertex is always projected onto a moving three-point
plane inside the differentiable map. Thus the selected tetra-volume is zero,
and its two triangulation mass polynomials agree along this manifold.
Other hull changes remain possible; every step rebuilds its actual geometry.
Floating nomination only; round the raw parameters and project rationally
before using the exhaustive exact V audit on an endpoint.
"""
from __future__ import annotations
import sys,json,time
from pathlib import Path

from scipy.optimize import minimize

from jointwide123 import widest_tree
from barflow123 import bar_structure,dual_edges,depths

def project(raw,indices):
 a,b,c,k=indices;n=torch.linalg.cross(raw[b]-raw[a],raw[c]-raw[a]);nn=(n*n).sum()
 z=raw[k]-((raw[k]-raw[a])*n).sum()/nn*n
 return raw.index_copy(0,torch.tensor([k]),z.reshape(1,3))

def values(raw,indices,derivatives=False):
 r=torch.tensor(raw);x=project(r,indices).detach().numpy();tr,fa,vi,hull=triangulate(x)

  a,b,c,k=indices;n=torch.linalg.cross(xx[b]-xx[a],xx[c]-xx[a]);n=n/torch.linalg.vector_norm(n)
  supporters=np.array([i for i in range(len(x)) if i not in indices]);support=-((xx[supporters]-xx[a])*n).sum(1)
  return torch.cat((reference_tree_guards(pp,root,parents),depths(qq,bs),vv.reshape(1),support))

  direction=sub.x[:-1].reshape(-1,3);ok=False

   alpha=2.**(-ls);rr=raw.copy();rr[a['ids']]+=alpha*direction
   try:aa=values(rr,indices)
   except Exception:continue

  trace.append(row)
  if it%10==0 or not ok:print(json.dumps(row),flush=True)
  if not ok:break
