```python
        poles.append(n/den)
    return z,v,poles,g,sig
```

## hybrid_reference_locator.py
```python
import sys,json
import numpy as np
from scipy.optimize import minimize
from h5_hybrid_explore import make,stat
K0=(7024915/1e6)*(6832329/1e6)/81

def setup(theta):
 z,v,q,g,s=make(theta,False)
 v=np.array(v,dtype=float)[:,:,0];q=np.array(q,dtype=float)[:,:,0];s=np.array(s,dtype=float)
 w,U=np.linalg.eigh(s); sr=(U*np.sqrt(w))@U.T
 return v@np.linalg.inv(sr),q@sr,sr

def metric(t):
 a,b,c,d,e=t
 B=np.array([[np.exp(a),0.,0.],[c,np.exp(b),0.],[d,e,np.exp(-a-b)]])
 return B@B.T

def loss(t,V,Q,detail=False):
 C=metric(t);Ci=np.linalg.inv(C)
 rr=np.einsum('ij,jk,ik->i',V,Ci,V)
 ll=np.einsum('ij,jk,ik->i',Q,C,Q)
 G=rr.max()*ll.max();K=max(1,G*G*K0)
 chi=np.maximum(0,rr[7:]*ll[0]/(V[7:]@Q[0])**2-1)
 L=((K+1)*np.sqrt(1+chi)+(K-1)*np.sqrt(chi))**2/(4*K)
 return (L.max(),G,chi,K) if detail else L.max()
if __name__=='__main__':
 out=[]
 for l in [.07,.08,.0825,.09,.1,.12,.15,.2,.3,.4]:
  th=[.15,.15,.15,.45,l];V,Q,sr=setup(th)
  res=minimize(loss,np.zeros(5),args=(V,Q),method='Powell',bounds=[(-2,2)]*5,options={'maxiter':1000,'xtol':1e-9,'ftol':1e-10})
  a=stat(th);vals=loss(res.x,V,Q,True)
  obj=dict(theta=th,reference=(sr@metric(res.x)@sr).tolist(),parameters=res.x.tolist(),loss=vals[0],Gamma=vals[1],chi=vals[2].tolist(),K=vals[3],need=a['need'],alpha_max=a['alpha_max'])
  out.append(obj);print(json.dumps(obj),flush=True)
 open('/mnt/data/hybrid_reference_locators.json','w').write(json.dumps(out,indent=2)+'\n')
```
