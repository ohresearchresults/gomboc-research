"""Finite joint actual-mass shape/metric scout on new C2 polar trees.
Uses full true-edge nominations from dual triangle adjacency, with coplanar
triangles merged by nominated primal coordinates. Exact audit is separate.
"""
from __future__ import annotations
import sys,json
from pathlib import Path
import numpy as np
from scipy.spatial import ConvexHull,QhullError
from scipy.optimize import differential_evolution
from twist113 import generate
from metric113 import metric,angles
from geometry109 import mass,from_h,census
ROOT=Path(__file__).resolve().parent

def free_generate(x,depth,mode):
    # Keep the original finite generator; remove only its unnecessary reference
    # centre-height restriction. Its validation is applied to fraction1 first.
    xx=np.asarray(x).copy();fraction=xx[5];xx[5]=.9999
    q,data=generate(xx,depth,mode)
    q[1:]*=fraction/.9999
    data['d']*=fraction/.9999
    return q,data

def geometry(q):
    dh=ConvexHull(q)
    if max(dh.equations[:,3])>=-1e-9:raise ValueError('unbounded')
    v0=np.linalg.solve(q[dh.simplices],np.ones((len(dh.simplices),3,1)))[:,:,0]
    _,ind,mapping=np.unique(np.round(v0,10),axis=0,return_index=True,return_inverse=True)
    v=v0[ind]
    pairs=np.array([(i,int(j)) for i,js in enumerate(dh.neighbors) for j in js if i<j and mapping[i]!=mapping[j]],int)
    pe=np.unique(np.sort(mapping[pairs],axis=1),axis=0)
    qe=np.array([sorted(set(dh.simplices[i])&set(dh.simplices[j])) for i,j in pairs],int)
    qe=np.unique(qe,axis=0)
    if len(v)-len(pe)+len(dh.vertices)!=2:raise ValueError('Euler')
    if np.min(np.linalg.norm(v[pe[:,0]]-v[pe[:,1]],axis=1))<1e-8:raise ValueError('tiny edge')
    vol,g,_=mass(v)
    return v,pe,qe,dh.vertices,vol,g

"""General (symmetry-breaking) affine continuation of new literal rational bodies.
Every metric uses the actual homogeneous centroid of that literal body; no free
reference point, vertex deformation or inertia surrogate is used.
"""
from __future__ import annotations
from fractions import Fraction as F
from pathlib import Path
import sys,json
import numpy as np
from scipy.optimize import differential_evolution
from metric113 import angles
from geometry109 import from_h,census
ROOT=Path(__file__).resolve().parent

def affine(t):
    a,d,b,c,e=t
    return np.array([[np.exp(a),b,c],[0.,np.exp(d),e],[0.,0.,1.]])

def run():
    out=[]
    for name in ('C2POLAR_M0_D14_G6','C2POLAR_M0_D20_G6','C2POLAR_M1_D20_G6'):
        exact=json.loads((ROOT/(name+'.audit.json')).read_text())
        v=np.array([[float(F(z)) for z in x] for x in exact['raw_vertices']]);g=np.array([float(F(z)) for z in exact['centroid']]);p=v-g
        q0=np.array(exact['input_N'],float)/exact['input_D'];q=q0/(1-q0@g)[:,None]
        pe=np.array(exact['edges']);active=np.array(exact['active_input_faces']);qe=active[np.array(exact['polar_edges'])]
        up=exact['strict_carriers'][1][0];sp=0;nonP=np.array([i for i in range(len(p)) if i!=up]);nonQ=active[active!=sp]
        calls=0;best=None
        def objective(x):
            nonlocal calls,best
            calls+=1;A=affine(x);P=p@A.T;Q=q@np.linalg.inv(A)
            cp=angles(P,pe)[nonP];cq=angles(Q,qe)[nonQ]
            ap=max(0.,1e-6-float(cp.min()));aq=max(0.,1e-6-float(cq.min()))
            rrP=np.einsum('ij,ij->i',P,P);rrQ=np.einsum('ij,ij->i',Q,Q)
            npv=max(0.,float(max(rrP[nonP])/rrP[up]-1)+1e-6);nqv=max(0.,float(max(rrQ[nonQ])/rrQ[sp]-1)+1e-6)
            val=max(ap,aq,npv,nqv)+.05*(ap+aq+npv+nqv)

                best=dict(x=x.tolist(),A=A.tolist(),loss=val,angle_violations=[ap,aq],norm_violations=[npv,nqv])
            return val
        opt=differential_evolution(objective,[(-5.,1.),(-5.,1.),(-1.,1.),(-3.,3.),(-3.,3.)],
            popsize=16,maxiter=500,seed=1113+len(p),tol=1e-9,atol=1e-12,polish=True,x0=np.zeros(5),workers=1,updating='immediate')

        row=dict(parent=name,selected_U=up,calls=calls,best=best,optimizer_success=bool(opt.success),message=opt.message,q=newq.tolist(),own=own,numerical_only=True)
        out.append(row);(ROOT/'FULLMETRIC113.json').write_text(json.dumps(out,indent=2)+'\n')
        print(name,'loss',best['loss'],'SU',len(own['S']),len(own['U']),'P',[x['depth'] for x in own['P']],'Q',[x['depth'] for x in own['Q']],flush=True)
    print('FULLMETRIC113_FINISHED',len(out))
if __name__=='__main__':run()
