def geometry(path=INPUT):

    Q=np.array([[int(x) for x in l.split()] for l in lines[1:]],float)/D
    assert Q.shape==(n,3)
    hull=ConvexHull(Q)
    assert len(hull.vertices)==n and np.max(hull.equations[:,-1])<0
    triples=sorted(tuple(sorted(int(i) for i in t)) for t in hull.simplices)
    X=np.array([np.linalg.solve(Q[list(t)], np.ones(3)) for t in triples])
    assert np.max(Q@X.T-1)<1e-9
    facets=[]; mass=0.; moment=np.zeros(3)
    for k,q in enumerate(Q):
        ids=[i for i,t in enumerate(triples) if k in t]
        c=X[ids].mean(axis=0); u=X[ids[0]]-c; u/=np.linalg.norm(u)
        v=np.cross(q,u);v/=np.linalg.norm(v)
        ids=sorted(ids,key=lambda i:np.arctan2((X[i]-c)@v,(X[i]-c)@u))
        facets.append(ids)
        for i in range(1,len(ids)-1):
            a,b,c=X[[ids[0],ids[i],ids[i+1]]]
            vol=np.linalg.det(np.array([a,b,c]))/6
            assert vol>0
            mass+=vol;moment+=vol*(a+b+c)/4
    g=moment/mass
    edges=sorted({tuple(sorted((ids[i],ids[(i+1)%len(ids)]))) for ids in facets for i in range(len(ids))})

def matrices(theta):

        pick=np.array([ix[np.argmax(c[ix])] for ix in self.groups]); best=c[pick]; grad=deriv[pick]

#!/usr/bin/env python3
"""Numerical finite affine descent of a genuine one-sided state's full bar.
Fixed allowed roots are an explicit search policy, not a geometric necessity.
"""
import json,argparse
from pathlib import Path
import numpy as np
from scipy.optimize import minimize
from metric_search import geometry,Landscape,matrices

def bars(X,edges,G,DG):
    birth=np.einsum('ni,ij,nj->n',X,G,X)
    db=np.einsum('ni,kij,nj->nk',X,DG,X)
    events=[]
    for i,j in edges:
        a=X[i];d=X[j]-a;t=float(np.clip(-(a@G@d)/(d@G@d),0,1));s=a+t*d
        if t==0:h=birth[i];dh=db[i]
        elif t==1:h=birth[j];dh=db[j]
        else:h=s@G@s;dh=np.einsum('i,kij,j->k',s,DG,s)
        events.append((float(h),i,j,dh))
    events.sort(key=lambda e:(-e[0],e[1],e[2]))
    parent=list(range(len(X)))
    def find(i):
        while parent[i]!=i:parent[i]=parent[parent[i]];i=parent[i]
        return i
    result=[]
    for h,i,j,dh in events:
        u=find(i);v=find(j)
        if u==v:continue
        if (birth[u],-u)<(birth[v],-v):u,v=v,u
        depth=1-h/birth[v]
        if depth>0:result.append(dict(depth=float(depth),birth=float(birth[v]),death=h,root=v,edge=[i,j],gradient=(h*db[v]/birth[v]**2-dh/birth[v]).tolist()))
        parent[v]=u
    return sorted(result,key=lambda r:-r['depth'])

class Constrained:
    def __init__(self,geo,eps):
        self.geo=geo;self.X=geo['P'];self.edges=geo['edges'];self.eps=eps
        self.pr=Landscape(geo['P'],geo['edges']);self.po=Landscape(geo['Y'],geo['pedges'])
        A,G,DG,J,DJ=matrices(np.zeros(5));p=self.pr.eval(G,DG);q=self.po.eval(J,DJ)
        self.uroots=np.where(p[0]<0)[0];self.sroots=np.where(q[0]<0)[0]
        self.umask=np.ones(len(geo['P']),bool);self.umask[self.uroots]=False
        self.smask=np.ones(len(geo['Y']),bool);self.smask[self.sroots]=False
    def objective(self,x):
        A,G,DG,J,DJ=matrices(x);bb=bars(self.X,self.edges,G,DG)
        if not bb:return 0.,np.zeros(5)
        return bb[0]['depth'],np.array(bb[0]['gradient'])
    def constraints(self,x,jac=False):
        A,G,DG,J,DJ=matrices(x);p=self.pr.eval(G,DG);q=self.po.eval(J,DJ)
        if jac:return np.vstack((p[1][self.umask],q[1][self.smask],-p[1][self.uroots],-q[1][self.sroots]))
        return np.r_[p[0][self.umask],q[0][self.smask],-p[0][self.uroots],-q[0][self.sroots]]-self.eps
    def report(self,x):
        A,G,DG,J,DJ=matrices(x);p=self.pr.eval(G,DG);q=self.po.eval(J,DJ)
        return dict(theta=x.tolist(),A=A.tolist(),S=np.where(q[0]<0)[0].tolist(),U=np.where(p[0]<0)[0].tolist(),constraints_min=float(self.constraints(x).min()),P=bars(self.X,self.edges,G,DG),Q=bars(self.geo['Y'],self.geo['pedges'],J,DJ),eig=np.linalg.eigvalsh(G).tolist())

def main():
    ap=argparse.ArgumentParser();ap.add_argument('--eps',type=float,default=1e-7);ap.add_argument('--starts',type=int,default=4);ap.add_argument('--bound',type=float,default=2);ap.add_argument('--maxiter',type=int,default=500);ap.add_argument('--seed',type=int,default=280104);ap.add_argument('--warm');ap.add_argument('--output',default='class_affine.json');a=ap.parse_args()

                    r=audit(Q[keep]);r['removed']=list(rm);r['S_input_original']=[keep[i] for i in r['S']];r['U_input_original']=[[keep[i] for i in tri] for tri in r['U']]
                except (ValueError,QhullError,np.linalg.LinAlgError) as e:r=dict(removed=list(rm),failure=str(e))
                reports.append(r)
                if 'S' in r and len(r['S'])==1:
                    paths.append(r);nq.append(rm);print('S1',rm,'U',len(r['U']),'E',r['E'],flush=True)
                    if len(r['U'])==1:targets.append(r)
                if len(reports)>=args.maxstates:break
            if len(reports)>=args.maxstates:break
        queue=nq
        print('DEPTH',depth,'NEXT',len(queue),'TOTAL',len(reports),flush=True)
        Path(args.output).write_text(json.dumps(dict(config=vars(args),targets=targets,paths=paths,reports=reports),indent=2))
        if not queue or targets or len(reports)>=args.maxstates:break
    print('TERMINAL','targets',len(targets),'S1',len(paths),'tested',len(reports),flush=True)
