"""Numerical nomination and rational point-hull integration for J141.
Fractions certify nominated simplicial facets; float routines are not mechanics proofs.
"""
import os
os.environ.setdefault('OPENBLAS_NUM_THREADS','1')
os.environ.setdefault('OMP_NUM_THREADS','1')
import numpy as np
from scipy.spatial import ConvexHull
from fractions import Fraction as Q

def dot(x,y):return sum(a*b for a,b in zip(x,y))
def cross(x,y):return [x[1]*y[2]-x[2]*y[1],x[2]*y[0]-x[0]*y[2],x[0]*y[1]-x[1]*y[0]]
def sub(x,y):return [a-b for a,b in zip(x,y)]

def hull(X):
 X=np.asarray(X,float)
 H=ConvexHull(X)
 T=H.simplices.copy()
 p=X[T]
 neg=np.einsum('ij,ij->i',np.cross(p[:,1]-p[:,0],p[:,2]-p[:,0]),H.equations[:,:3])<0
 T[neg]=T[neg][:,[0,2,1]]
 p=X[T];d=np.einsum('ij,ij->i',p[:,0],np.cross(p[:,1],p[:,2]))
 V=d.sum()/6
 if V<=0:raise ValueError('Nonpositive volume')
 S=p.sum(1)
 M=np.einsum('i,ij->j',d,S)/24
 J=(np.einsum('i,ij,ik->jk',d,S,S)+np.einsum('i,iaj,iak->jk',d,p,p))/120
 return dict(X=X,tri=T,active=H.vertices,V=V,g=M/V,second=J/V,planes=H.equations)

def exact_hull(X):
 """Complete triangulated boundary candidate, all exact supporting inequalities.
 Requires simplicial hull; no inference of non-simplicial true-edge counts.
 """
 X=[[Q(a) for a in x] for x in X];hf=hull(np.array(X,float));T=hf['tri']
 V=Q(0);M=[Q(0)]*3;F=[];E={};normal_weights=[[Q(0)]*3 for x in X]
 for tr in T:
  i,j,k=map(int,tr);p,q,r=X[i],X[j],X[k]
  n=cross(sub(q,p),sub(r,p));h=dot(n,p)
  sl=[h-dot(n,x) for x in X]
  assert min(sl)>=0 and sum(v==0 for v in sl)==3,(tr,'non-simplicial or missing facet')
  det=dot(p,cross(q,r));V+=det/6
  for z in range(3):M[z]+=det*(p[z]+q[z]+r[z])/24
  for a,b in ((i,j),(j,k),(k,i)):
   key=tuple(sorted((a,b)));E.setdefault(key,[]).append((a,b))
  for a in (i,j,k):
   for z in range(3):normal_weights[a][z]+=n[z]/6
  F.append((i,j,k))
 assert all(len(v)==2 and v[0]==v[1][::-1] for v in E.values())
 assert len(set(a for f in F for a in f))-len(E)+len(F)==2
 assert V>0
 g=[a/V for a in M]
 nu=[[a/V for a in n] for n in normal_weights]
 return dict(X=X,faces=F,edges=list(E),V=V,g=g,nu=nu)

def potentials(X,root):
 """Maximum-product closure for unnormalized rays, normalized a_root=1."""
 X=np.asarray(X,float);C=np.maximum(0,X@X.T/(X*X).sum(1)[:,None]);np.fill_diagonal(C,0)
 p=np.zeros(len(X));p[root]=1
 for z in range(len(X)):
  q=np.maximum(p,(C*p[None,:]).max(1));q[root]=1
  if np.max(abs(q-p))<1e-14:p=q;break
  p=q
 else:raise ValueError('potential closure did not stabilize')
 if min(p)<=0:raise ValueError('unreachable root')
 return p,C

Floating feasibility/infeasibility is nomination, not exact certification.

def joint_lp(X,eps=0):

 # sum nu_i=0
 for k in range(3):
  row=np.zeros((n,3));row[:,k]=1;A.append(row.ravel());rhs.append(0.)
 # sum x_i nu_i^T=I
 for k in range(3):
  for l in range(3):
   row=np.zeros((n,3));row[:,l]=X[:,k];A.append(row.ravel());rhs.append(float(k==l))
 # sum (x_i dot nu_i) x_i=0

"""Floating phase-I dual nomination; output has no exact-proof status.
Uses NumPy 2.3.5 / SciPy 1.17.0, HiGHS defaults, eps=0.
"""
import json
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix,hstack
from scipy.linalg import qr

def problem(X):
 n=len(X); A=[];e=[]
 for k in range(3):
  row=np.zeros((n,3));row[:,k]=1;A.append(row.ravel());e.append(0.)
 for k in range(3):
  for l in range(3):
   row=np.zeros((n,3));row[:,l]=X[:,k];A.append(row.ravel());e.append(float(k==l))
 for k in range(3):
  row=X[:,k,None]*X;A.append(row.ravel());e.append(0.)
 B=lil_matrix((2*n*(n-1)+n,3*n));b=[];meta=[];r=0
 for i in range(n):
  for j in range(n):
   if i==j:continue
   for sign in [-1,1]:
    B[r,3*j:3*j+3]=sign*(X[j]-X[i]);b.append(float(sign==1));meta.append([i,j,sign]);r+=1
 for j in range(n):B[r,3*j:3*j+3]=-X[j];b.append(0.);meta.append([-1,j,-1]);r+=1
 return np.array(A),np.array(e),B.tocsr(),np.array(b),meta

def main():
 src=json.load(open('JOINT141_FLOAT.json'));out={}
 for root,o in src.items():
  X=np.array(o['X']);n=len(X);A,e,B,b,meta=problem(X)
  # Minimize a common violation tau >=0. All affine equalities remain exact in the numerical LP.
  z=linprog(np.r_[np.zeros(3*n),1.],A_ub=hstack((B,-np.ones((len(b),1)))).tocsr(),b_ub=b,
            A_eq=np.c_[A,np.zeros(len(A))],b_eq=e,bounds=[(None,None)]*(3*n)+[(0,None)],method='highs')
  assert z.success and z.fun>0
  lam=-z.ineqlin.marginals;mu=-z.eqlin.marginals
  D=10**12
  L=[[meta[i],int(round(v*D))]for i,v in enumerate(lam)if round(v*D)>0]
  M=[int(round(v*D))for v in mu]
  parents=[]
  for j in range(n):
   _,_,piv=qr((X[j]-X).T,pivoting=True);parents.append(list(map(int,piv[:3])))
  residual=B.T@lam+A.T@mu;q=float(b@lam+e@mu)
  out[root]=dict(denominator=D,lambda_rows=L,mu=M,bases=parents,nomination_tau=float(z.fun),nomination_rhs=q,
                nomination_max_residual=float(max(abs(residual))))
  print(root,'tau',z.fun,'rhs',q,'duals',len(L),len(M),'residual',max(abs(residual)),flush=True)
 with open('DUAL141_RECIPES.json','w')as f:json.dump(out,f,sort_keys=True,separators=(',',':'));f.write('\n')
