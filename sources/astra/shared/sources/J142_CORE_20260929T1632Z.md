def positive_affine(Z,D,F,lam,LD=10**9):
    total=Q(0)
    for f in tetrahedra(F):
        v=Q(det(*[Z[i]for i in f]),6*D**3);assert v>0
        hs=[Q(1)]+[1+Q(dot(lam,Z[i]),D*LD)for i in f]
        if min(hs)>=0:term=sum(hs)
        elif max(hs)<=0:term=Q(0)
        else:

                    if j!=i:d*=h-k
                term+=h**4/d
        assert term>=0
        total+=v*term/4
    return total
